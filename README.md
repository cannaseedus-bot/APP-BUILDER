### Usage Example

```python
from checkpoint_manager import (
    MX2LMCheckpointManager,
    CheckpointMeta,
    ASXRAMSnapshot,
    RLHFMetrics,
    XCFEVectors,
    PolyglotRuntimeState
)

# Initialize checkpoint manager
manager = MX2LMCheckpointManager('./checkpoints')

# Create checkpoint with polyglot state
meta = CheckpointMeta(
    step=1000,
    epoch=1,
    loss=0.5,
    best=True,
    asx_ram=ASXRAMSnapshot(
        ngrams_count=1000,
        bigrams_count=500,
        # ... other n-gram types
    ),
    xcfe=XCFEVectors(
        language_active="KUHUL",
        pipeline_stage="chen"
    ),
    polyglot=PolyglotRuntimeState(
        kuhul_active=True,
        seal_kuhul_glow=1.0
    )
)

# Save checkpoint
manager.save_checkpoint(model, optimizer, meta)

# Load checkpoint
meta = manager.load_best_or_last()
```
