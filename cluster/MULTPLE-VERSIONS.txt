  async scrapeForTraining(agentId, urls, dataSchema) {

    for (const url of urls) {
      const result = await this.fetchForAgent(agentId, url);
      
      // Parse based on data schema
      const parsed = await this.parseWithSchema(result.data, dataSchema);
      scrapedData.push(parsed);
      
      // Rate limiting
      await new Promise(resolve => setTimeout(resolve, 1000));
    }
    
    return scrapedData;
  },
  
  async parseWithSchema(html, schema) {
    // Simple HTML parsing based on schema
    if (schema.fields) {
      const data = {};
      for (const field of schema.fields) {
        // Extract data based on field selectors
        data[field.name] = this.extractField(html, field);
      }
      return data;
    }
    
    return { raw: html.slice(0, 1000) + '...' }; // Truncate for demo
  },
  
  extractField(html, field) {
    // Simple field extraction (would use proper HTML parsing in real implementation)
    if (field.selector) {
      const regex = new RegExp(field.selector, 'i');
      const match = html.match(regex);
      return match ? match[1] : null;
    }
    return null;
  }
};

/* -------------------------------------------------------------------------
   AGENT ORCHESTRATION API
------------------------------------------------------------------------- */

async function respondAgentOrchestration(url) {
  const path = url.pathname.replace(/^\/api\/agents\//, '');
  const parts = path.split('/');
  
  // Spawn new agent
  if (parts[0] === 'spawn') {
    const { topic, requirements } = await url.json();
    const agent = await MicronautFactory.spawnTopicAgent(topic, requirements);
    
    return new Response(JSON.stringify({
      success: true,
      agent,
      message: `Spawned ${topic} micronaut with mini-OS runtime`
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  // Agent control
  if (parts[0] === 'control' && parts[1]) {
    const agentId = parts[1];
    const { action, data } = await url.json();
    
    let result;
    switch (action) {
      case 'ingest':
        result = await StaticTunnelBrowser.fetchForAgent(agentId, data.url);
        break;
      case 'train':
        result = await TrainingBotManager.spawnTrainingBot(data.dataset, data.modelConfig, agentId);
        break;
      case 'query':
        result = await this.queryAgent(agentId, data.query);
        break;
      default:
        result = { error: 'unknown_action', action };
    }
    
    return new Response(JSON.stringify({ agentId, action, result }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  // Training management
  if (parts[0] === 'training' && parts[1]) {
    const trainingId = parts[1];
    
    if (parts[2] === 'monitor') {
      const trainingBot = await TrainingBotManager.monitorTraining({ trainingId });
      return new Response(JSON.stringify({ trainingId, status: trainingBot }), {
        headers: { 'Content-Type': 'application/json' }
      });
    }
  }
  
  // Static tunnel browsing
  if (parts[0] === 'tunnel' && parts[1]) {
    const agentId = parts[1];
    const { url, options } = await url.json();
    
    const result = await StaticTunnelBrowser.fetchForAgent(agentId, url, options);
    return new Response(JSON.stringify(result), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  // List active agents
  if (parts[0] === 'list') {
    const activeAgents = Array.from(KuhulKernel.processes.entries())
      .filter(([pid, process]) => pid.startsWith('micronaut_'))
      .map(([pid, process]) => ({
        id: pid,
        topic: process.context?.topic,
        status: process.status,
        runtime: process.runtime
      }));
    
    return new Response(JSON.stringify({ activeAgents }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  return new Response(
    JSON.stringify({ error: 'agent_endpoint_not_found', path }),
    { status: 404, headers: { 'Content-Type': 'application/json' } }
  );
}

async function respondAgentOrchestration(url) {
  const path = url.pathname.replace(/^\/api\/agents\//, '');
  const parts = path.split('/');
  
  // Spawn new agent
  if (parts[0] === 'spawn') {
    const { topic, requirements } = await url.json();
    const agent = await MicronautFactory.spawnTopicAgent(topic, requirements);
    
    return new Response(JSON.stringify({
      success: true,
      agent,
      message: `Spawned ${topic} micronaut with mini-OS runtime`
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  // Agent control
  if (parts[0] === 'control' && parts[1]) {
    const agentId = parts[1];
    const { action, data } = await url.json();
    
    let result;
    switch (action) {
      case 'ingest':
        result = await StaticTunnelBrowser.fetchForAgent(agentId, data.url);
        break;
      case 'train':
        result = await TrainingBotManager.spawnTrainingBot(data.dataset, data.modelConfig, agentId);
        break;
      case 'query':
        result = await this.queryAgent(agentId, data.query);
        break;
      default:
        result = { error: 'unknown_action', action };
    }
    
    return new Response(JSON.stringify({ agentId, action, result }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  // Training management
  if (parts[0] === 'training' && parts[1]) {
    const trainingId = parts[1];
    
    if (parts[2] === 'monitor') {
      const trainingBot = await TrainingBotManager.monitorTraining({ trainingId });
      return new Response(JSON.stringify({ trainingId, status: trainingBot }), {
        headers: { 'Content-Type': 'application/json' }
      });
    }
  }
  
  // Static tunnel browsing
  if (parts[0] === 'tunnel' && parts[1]) {
    const agentId = parts[1];
    const { url, options } = await url.json();
    
    const result = await StaticTunnelBrowser.fetchForAgent(agentId, url, options);
    return new Response(JSON.stringify(result), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  // List active agents
  if (parts[0] === 'list') {
    const activeAgents = Array.from(KuhulKernel.processes.entries())
      .filter(([pid, process]) => pid.startsWith('micronaut_'))
      .map(([pid, process]) => ({
        id: pid,
        topic: process.context?.topic,
        status: process.status,
        runtime: process.runtime
      }));
    
    return new Response(JSON.stringify({ activeAgents }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
  
  return new Response(
    JSON.stringify({ error: 'agent_endpoint_not_found', path }),
    { status: 404, headers: { 'Content-Type': 'application/json' } }
  );
}
