XJSON Runtime Mesh

The xjsonRun helper sends a program + context to any node exposing /xjson/run. HIVE-NET simply fans this out across multiple servers. 

// xjson-runtime.js (client helper)
export async function xjsonRun(baseUrl, program, context = {}) {
  const res = await fetch(
    baseUrl.replace(/\/$/, "") + "/xjson/run",
    {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ program, context })
    }
  );

  if (!res.ok) throw new Error("XJSON request failed");

  const data = await res.json().catch(() => ({}));
  if (!data.ok) throw new Error("XJSON error: " + (data.error || "unknown"));
  return data.result;
}

XJSON is the **canonical backend**: KLH, K’uhul, Basher, HIVE-NET, tapes — all speak XJSON.

SCXQ2-Lite Compactor

SCXQ2 compresses text and training data into tiny glyph packets. For browser use, we can run a base64 + dictionary variant via JS. 

// SCXQ2-lite: browser-safe compactor
const SCXQ2 = {
  encode(str) {
    return btoa(unescape(encodeURIComponent(str)));
  },
  decode(payload) {
    try { return decodeURIComponent(escape(atob(payload))); }
    catch { return ""; }
  }
};

// Example: compress a mini XJSON program
const packed = SCXQ2.encode(JSON.stringify({
  node: "$url",
  status: "online"
}));

In the full engine, SCX adds **dictionary encoding + RLE + Huffman** for 75%+ savings on n-gram corpora and OS traces.

Seed & RLHF Flow

K’uhul treats every user correction as a **seed tape**: text + score → hash → broadcast to SW + seed stream + local n-gram builder. 

async function pushTrain(text, score) {
  const payload = { text, score, ts: Date.now() };
  await fetch("/api/train", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload)
  });

  const data = JSON.stringify(payload);
  const buf  = await crypto.subtle.digest("SHA-256",
              new TextEncoder().encode(data));
  const hash = [...new Uint8Array(buf)]
      .map(b => b.toString(16).padStart(2,"0")).join("");

  if (navigator.serviceWorker?.controller) {
    navigator.serviceWorker.controller.postMessage({
      type: "seed",
      payload: { hash, ts: Date.now() }
    });
  }
}
ASX-RAM Bridge

K’uhul uses **ASX-RAM** (SW message bus) instead of localStorage for shared brain state. 

const ASXRAM = {
  store(key, value) {
    if (!navigator.serviceWorker?.controller) return;
    navigator.serviceWorker.controller.postMessage({
      type: "ram-store",
      key,
      value
    });
  },
  get(key) {
    return new Promise(resolve => {
      if (!navigator.serviceWorker?.controller) return resolve(null);
      const handler = (e) => {
        if (e.data?.type === "ram-data" && e.data.key === key) {
          navigator.serviceWorker.removeEventListener("message", handler);
          resolve(e.data.value);
        }
      };
      navigator.serviceWorker.addEventListener("message", handler);
      navigator.serviceWorker.controller.postMessage({
        type: "ram-get",
        key
      });
    });
  }
};

ASX Bridge (Static Node)

asx-bridge.js is a zero-dependency Node HTTP server serving your MEGA CODEX and assets from a single root. 

// runtime/asx-bridge.js (core idea)
const http = require("http");
const fs   = require("fs");
const path = require("path");

const ROOT = path.join(__dirname, "..");
const PORT = process.env.PORT || 3000;

http.createServer((req, res) => {
  let pathname = decodeURIComponent(req.url || "/");
  if (pathname === "/") pathname = "/index.html";
  const filePath = path.join(ROOT, pathname);
  fs.readFile(filePath, (err, data) => {
    if (err) {
      res.writeHead(404, { "Content-Type": "text/plain" });
      return res.end("Not found");
    }
    res.writeHead(200);
    res.end(data);
  });
}).listen(PORT);

Each section in this MEGA CODEX is treated as a tape and broadcasts heartbeats to the service worker + KLH router. 

<script>
(function () {
  const TAPE_ID = "asx-tapes"; // this section

  function sendHeartbeat() {
    fetch("/api/local/" + encodeURIComponent(TAPE_ID)).catch(() => {});
    if (navigator.serviceWorker?.controller) {
      navigator.serviceWorker.controller.postMessage({
        type: "TAPE_HEARTBEAT",

        url: location.pathname,
